def gqa_attention(q, k, v, num_heads, num_kv_groups):
    # Q 头更多，K/V 头更少
    q = split_heads(q, num_heads)
    k = split_heads(k, num_kv_groups)
    v = split_heads(v, num_kv_groups)
    k = repeat_groups(k, num_heads // num_kv_groups)
    v = repeat_groups(v, num_heads // num_kv_groups)
    attn = softmax(q @ k.transpose(-2, -1))
    return merge_heads(attn @ v)

复杂度：MHA > GQA > MQA（K/V 头数减少）。

def flash_block(q, k, v, block_size=64):
    # 分块 + 在线 softmax
    for q_block in blocks(q):
        m, l, acc = init_stats(q_block)
        for k_block, v_block in blocks(k, v):
            scores = q_block @ k_block.T
            m, l, acc = online_softmax_update(scores, v_block, m, l, acc)
        yield acc / l

def rope_apply(q, k):
    # 旋转位置编码（简化）
    cos, sin = build_angles(q)
    q = rotate_pairs(q, cos, sin)
    k = rotate_pairs(k, cos, sin)
    return q, k

def alibi_bias(num_heads, seq_len):
    slopes = get_head_slopes(num_heads)
    pos = arange(seq_len)
    return slopes[:, None, None] * (pos[None, :] - pos[:, None])

def t5_relative_position_bias(num_heads, seq_len, num_buckets=32):
    rel = bucketize_relative_positions(seq_len, num_buckets)
    return bias_table[num_heads, num_buckets][..., rel]
  • RoPE：旋转矩阵注入位置信息，适合长上下文。
  • ALiBi：线性偏置，无需显式位置嵌入。
  • T5：bucket 化相对位置，更适合长序列泛化。

def diou(box1, box2):
    iou = iou(box1, box2)
    center_dist = ||c1 - c2||^2
    diag = ||c_enclosing||^2
    return iou - center_dist / diag
